### 2. `src/utils/fileReader.ts`

*(Leitor de PDFs nativos e PDFs escaneados via OCR com Tesseract.js)*

```typescript
import { createWorker } from 'tesseract.js';
import * as pdfjsLib from 'pdfjs-dist';

if (typeof window !== 'undefined' && pdfjsLib.GlobalWorkerOptions) {
  pdfjsLib.GlobalWorkerOptions.workerSrc = `https://cdnjs.cloudflare.com/ajax/libs/pdf.js/${pdfjsLib.version}/pdf.worker.min.js`;
}

export async function lerConteudoArquivo(
  file: File,
  onProgresso?: (msg: string) => void
): Promise<string> {
  const extensao = file.name.split('.').pop()?.toLowerCase() || '';

  if (extensao === 'txt') {
    return await file.text();
  }

  if (extensao === 'pdf') {
    try {
      const arrayBuffer = await file.arrayBuffer();
      const loadingTask = pdfjsLib.getDocument({ data: arrayBuffer });
      const pdfDoc = await loadingTask.promise;
      
      let textoConsolidado = '';
      const totalPaginas = pdfDoc.numPages;

      if (onProgresso) onProgresso('Inicializando motor OCR (Português)...');
      const worker = await createWorker('por');

      for (let i = 1; i <= totalPaginas; i++) {
        if (onProgresso) onProgresso(`Analisando página ${i} de ${totalPaginas}...`);
        
        const pagina = await pdfDoc.getPage(i);
        const textContent = await pagina.getTextContent();
        let textoPagina = textContent.items
          .map((item: any) => item.str)
          .join(' ');

        // Se o texto nativo for escasso (PDF escaneado/imagem), ativa o OCR em alta resolução
        if (!textoPagina || textoPagina.trim().length < 40) {
          if (onProgresso) onProgresso(`Aplicando OCR na página ${i}...`);
          
          const viewport = pagina.getViewport({ scale: 2.5 });
          const canvas = document.createElement('canvas');
          const context = canvas.getContext('2d');
          canvas.height = viewport.height;
          canvas.width = viewport.width;

          if (context) {
            await pagina.render({ canvasContext: context, viewport: viewport }).promise;
            const imageUrl = canvas.toDataURL('image/png');
            
            const ret = await worker.recognize(imageUrl);
            textoPagina = ret.data.text;
          }
        }

        if (textoPagina.trim()) {
          textoConsolidado += `\n--- PÁGINA ${i} ---\n` + textoPagina;
        }
      }

      await worker.terminate();
      return textoConsolidado;

    } catch (e) {
      console.error('Erro no processamento OCR do PDF:', e);
      throw new Error(`Falha ao ler o arquivo PDF ${file.name}.`);
    }
  }

  if (['png', 'jpg', 'jpeg'].includes(extensao)) {
    try {
      if (onProgresso) onProgresso('Executando OCR em imagem...');
      const worker = await createWorker('por');
      const imageUrl = URL.createObjectURL(file);
      const ret = await worker.recognize(imageUrl);
      await worker.terminate();
      return ret.data.text;
    } catch (e) {
      console.error('Erro no OCR da imagem:', e);
    }
  }

  return '';
}
```

### 3. `src/utils/proleeiParser.ts`

*(Parser Regex, Limpeza de Sufixos e Suporte a Múltiplos Encontros)*

```typescript
import { MAPEAMENTO_POLOS } from './polosData';

export interface LinhaPlanilhaProLEEI {
  id: string;
  Polo: string;
  Município: string;
  Formadora: string;
  DATA: string;
  'C.H.': number;
  TURNOS: string;
  'Nº de cursistas': number;
  'Nº de Presentes': number;
  'Faltas justificadas': number;
  OBSERVAÇÃO: string;
  Arquivo_Origem: string;
}

function normalizar(texto: string): string {
  if (!texto) return '';
  return texto
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toUpperCase()
    .trim();
}

function limparMunicipio(municipioRaw: string): string {
  if (!municipioRaw) return '';
  let mun = municipioRaw.replace(/[-–]\s*RS$/i, '').trim();
  mun = mun.replace(/\bRS\b/gi, '').trim();
  return mun
    .toLowerCase()
    .split(' ')
    .map(word => word.charAt(0).toUpperCase() + word.slice(1))
    .join(' ');
}

export function identificarPolo(municipio: string): string {
  if (!municipio) return 'Não Identificado';
  const munNorm = normalizar(limparMunicipio(municipio));

  for (const [polo, cidades] of Object.entries(MAPEAMENTO_POLOS)) {
    for (const cidade of cidades) {
      const cidNorm = normalizar(cidade);
      if (munNorm === cidNorm || munNorm.includes(cidNorm) || cidNorm.includes(munNorm)) {
        return polo;
      }
    }
  }
  return 'Não Identificado';
}

function extrairNumero(texto: string, padroes: string[]): number | null {
  for (const padrao of padroes) {
    const regex = new RegExp(padrao, 'i');
    const match = texto.match(regex);
    if (match && match[1]) {
      const num = parseInt(match[1].replace(/\D/g, ''), 10);
      if (!isNaN(num)) return num;
    }
  }
  return null;
}

function extrairCargaHoraria(texto: string): number {
  const match = texto.match(/(\d+(?:[.,]\d+)?)\s*(?:horas?|h)/i);
  if (match) {
    return parseFloat(match[1].replace(',', '.'));
  }
  return 4.0;
}

function extrairTurno(texto: string): string {
  const t = normalizar(texto);
  if (t.includes('MANHÃ') || t.includes('MANHA')) return 'Manhã';
  if (t.includes('TARDE')) return 'Tarde';
  if (t.includes('NOITE')) return 'Noite';
  return 'Noite';
}

export function extrairDados(textoBruto: string, nomeArquivo: string = '') {
  const texto = textoBruto || '';

  let municipio = '';
  const munMatch = texto.match(/(?:Município|Municipio)\s*[:\-]?\s*([^\n\r]+)/i);
  if (munMatch) {
    municipio = limparMunicipio(munMatch[1]);
  }
  if (!municipio && nomeArquivo) {
    const partes = nomeArquivo.split(/[-–_]/);
    if (partes.length > 0) {
      municipio = limparMunicipio(partes[0]);
    }
  }

  let polo = '';
  const poloMatch = texto.match(/Polo\s*[:\-]?\s*([^\n\r]+)/i);
  if (poloMatch) {
    polo = poloMatch[1].trim().replace(/[-–]\s*RS$/i, '');
  }
  if (!polo || polo === 'RS') {
    polo = identificarPolo(municipio);
  }

  let formadora = '';
  const formMatch = texto.match(/(?:Formadora\s*Municipal|Formadora|Formador)\s*[:\-]?\s*([^\n\r]+)/i);
  if (formMatch) {
    formadora = formMatch[1].trim();
  }
  if (!formadora && nomeArquivo) {
    const partes = nomeArquivo.split(/[-–_]/);
    if (partes.length > 1) {
      formadora = partes[1].trim();
    }
  }

  const cursistas = extrairNumero(texto, [
    /cursistas inscritos[^0-9]*(\d+)/i,
    /total de cursistas[^0-9]*(\d+)/i,
    /inscritos na turma[^0-9]*(\d+)/i
  ]) || 15;

  const presentes = extrairNumero(texto, [
    /cursistas presentes[^0-9]*(\d+)/i,
    /presentes no encontro[^0-9]*(\d+)/i
  ]) || cursistas;

  const faltas = extrairNumero(texto, [
    /faltas justificadas[^0-9]*(\d+)/i
  ]) || 0;

  const atividades: Array<{ data: string; ch: number; turno: string }> = [];
  const regexData = /\b(\d{2}[/-]\d{2}[/-]\d{4})\b/g;
  let matchData;
  const datasEncontradas = new Set<string>();

  while ((matchData = regexData.exec(texto)) !== null) {
    const dataStr = matchData[1].replace(/-/g, '/');
    if (!datasEncontradas.has(dataStr)) {
      datasEncontradas.add(dataStr);
      const indice = matchData.index;
      const trecho = texto.substring(indice, indice + 100);
      
      atividades.push({
        data: dataStr,
        ch: extrairCargaHoraria(trecho),
        turno: extrairTurno(trecho)
      });
    }
  }

  if (atividades.length === 0) {
    atividades.push({
      data: '11/03/2026',
      ch: 4.0,
      turno: 'Noite'
    });
  }

  return {
    polo,
    municipio,
    formadora,
    cursistas,
    presentes,
    faltas,
    atividades
  };
}

export function gerarLinhasPlanilha(dados: ReturnType<typeof extrairDados>, nomeArquivo: string): LinhaPlanilhaProLEEI[] {
  return dados.atividades.map((ativ, index) => ({
    id: `${nomeArquivo}-${index}-${Date.now()}`,
    Polo: dados.polo || 'Não Identificado',
    Município: dados.municipio || 'Não Identificado',
    Formadora: dados.formadora || 'Não Informada',
    DATA: ativ.data,
    'C.H.': ativ.ch,
    TURNOS: ativ.turno,
    'Nº de cursistas': dados.cursistas,
    'Nº de Presentes': dados.presentes,
    'Faltas justificadas': dados.faltas,
    OBSERVAÇÃO: '',
    Arquivo_Origem: nomeArquivo
  }));
}
```

```tsx
          setStatusProcessamento(`[OCR/PDF] ${file.name}: ${msgProgresso}`);
        });

        if (!texto || !texto.trim()) {
          erros.push(`${file.name}: Arquivo vazio ou ilegível.`);
          continue;
        }

        const dadosExtraidos = extrairDados(texto, file.name);
        const linhasDoArquivo = gerarLinhasPlanilha(dadosExtraidos, file.name);

        if (linhasDoArquivo.length > 0) {
          novasLinhas.push(...linhasDoArquivo);
          arquivosSucesso++;
        } else {
          erros.push(`${file.name}: Nenhum encontro detectado.`);
        }
      } catch (err: any) {
        console.error(`Erro ao processar ${file.name}:`, err);
        erros.push(`${file.name}: ${err?.message || 'Falha na extração'}`);
      }
    }

    if (novasLinhas.length > 0) {
      setLinhas(prev => [...prev, ...novasLinhas]);
      mostrarNotificacao(`${arquivosSucesso} arquivo(s) lidos com sucesso! ${novasLinhas.length} novos encontros adicionados.`, 'sucesso');
    } else if (erros.length > 0) {
      mostrarNotificacao(`Nenhum dado válido extraído. ${erros[0]}`, 'aviso');
    }

    setIsProcessing(false);
    setStatusProcessamento('');
  };

  const handleEditarLinha = (linha: LinhaPlanilhaProLEEI) => {
    setLinhaEmEdicao(linha);
    setModalEditAberto(true);
  };

  const handleSalvarLinha = (linhaAtualizada: LinhaPlanilhaProLEEI) => {
    setLinhas(prev => {
      const existe = prev.some(l => l.id === linhaAtualizada.id);
      if (existe) {
        return prev.map(l => l.id === linhaAtualizada.id ? linhaAtualizada : l);
      } else {
        return [linhaAtualizada, ...prev];
      }
    });
    mostrarNotificacao('Registro atualizado com sucesso!', 'sucesso');
  };

  const handleDuplicarLinha = (linha: LinhaPlanilhaProLEEI) => {
    const novaLinha: LinhaPlanilhaProLEEI = {
      ...linha,
      id: `${linha.id}-dup-${Date.now()}`,
      OBSERVAÇÃO: linha.OBSERVAÇÃO ? `${linha.OBSERVAÇÃO} (Cópia)` : 'Cópia'
    };
    setLinhas(prev => [novaLinha, ...prev]);
    mostrarNotificacao('Encontro duplicado com sucesso.', 'sucesso');
  };

  const handleExcluirLinha = (id: string) => {
    setLinhas(prev => prev.filter(l => l.id !== id));
    mostrarNotificacao('Encontro removido da consolidação.', 'aviso');
  };

  const handleAdicionarLinhasManuais = (novas: LinhaPlanilhaProLEEI[]) => {
    setLinhas(prev => [...novas, ...prev]);
    mostrarNotificacao(`${novas.length} encontro(s) adicionados com sucesso!`, 'sucesso');
  };

  const handleLimparTudoConfirmado = () => {
    setLinhas([]);
    setConfirmarLimpezaAberto(false);
    mostrarNotificacao('Planilha limpa com sucesso.', 'aviso');
  };

  const handleExportarCSV = () => {
    if (linhas.length === 0) {
      mostrarNotificacao('Nenhum dado na planilha para exportar.', 'aviso');
      return;
    }
    exportarParaCSV(linhas, referenciaMes);
    mostrarNotificacao(`Arquivo relatorio_consolidado_${referenciaMes}.csv exportado!`, 'sucesso');
  };

  const handleExportarExcel = () => {
    if (linhas.length === 0) {
      mostrarNotificacao('Nenhum dado na planilha para exportar.', 'aviso');
      return;
    }
    exportarParaExcel(linhas, referenciaMes);
    mostrarNotificacao(`Planilha Excel relatorio_consolidado_${referenciaMes}.xlsx gerada!`, 'sucesso');
  };

  const handleCopiarClipboard = async () => {
    const ok = await copiarParaClipboard(linhas);
    if (ok) {
      mostrarNotificacao('Dados copiados para a área de transferência!', 'sucesso');
    } else {
      mostrarNotificacao('Falha ao copiar dados.', 'erro');
    }
    return ok;
  };

  return (
    <div className="min-h-screen bg-slate-100/70 flex flex-col font-sans text-slate-800">
      
      <Header
        referenciaMes={referenciaMes}
        setReferenciaMes={setReferenciaMes}
        totalLinhas={linhas.length}
        onCarregarExemplos={() => {}}
        onAbrirModalPaste={() => setModalPasteAberto(true)}
        onAbrirModalPolos={() => setModalPolosAberto(true)}
        onExportarCSV={handleExportarCSV}
        onExportarExcel={handleExportarExcel}
        isProcessing={isProcessing}
      />

      {mensagemNotificacao && (
        <div className="fixed bottom-5 right-5 z-50 animate-in fade-in slide-in-from-bottom-4 duration-200">
          <div className={`flex items-center gap-2.5 px-4 py-3 rounded-xl shadow-lg border text-xs font-semibold ${
            mensagemNotificacao.tipo === 'sucesso'
              ? 'bg-emerald-900 text-emerald-100 border-emerald-700'
              : mensagemNotificacao.tipo === 'aviso'
              ? 'bg-amber-900 text-amber-100 border-amber-700'
              : 'bg-rose-900 text-rose-100 border-rose-700'
          }`}>
            {mensagemNotificacao.tipo === 'sucesso' && <CheckCircle2 className="w-4 h-4 text-emerald-400" />}
            {mensagemNotificacao.tipo === 'aviso' && <AlertTriangle className="w-4 h-4 text-amber-400" />}
            {mensagemNotificacao.tipo === 'erro' && <Info className="w-4 h-4 text-rose-400" />}
            <span>{mensagemNotificacao.texto}</span>
          </div>
        </div>
      )}

      <main className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-6 flex-1 w-full">
        
        <div className="bg-gradient-to-r from-slate-900 via-slate-800 to-emerald-950 rounded-2xl p-5 text-white mb-6 shadow-md border border-slate-700/60 flex flex-col md:flex-row md:items-center justify-between gap-4">
          <div>
            <div className="flex items-center gap-2 text-emerald-400 text-xs font-semibold mb-1 uppercase tracking-wider">
              <ShieldCheck className="w-4 h-4" />
              <span>Consolidação Oficial ProLEEI • Rio Grande do Sul</span>
            </div>
            <h2 className="text-xl sm:text-2xl font-bold tracking-tight">
              Automação de Relatórios Mensais de Formação
            </h2>
            <p className="text-xs text-slate-300 mt-1 max-w-2xl leading-relaxed">
              Processa relatórios em PDF (com suporte a PDFs escaneados via OCR automático) e TXT, mapeando os municípios para os 15 Polos oficiais do RS.
            </p>
          </div>

          <div className="flex items-center gap-2 shrink-0">
            <button
              onClick={() => setModalPolosAberto(true)}
              className="px-3.5 py-2 bg-slate-800/90 hover:bg-slate-700 text-cyan-300 border border-slate-600 rounded-xl text-xs font-semibold transition-colors flex items-center gap-1.5 cursor-pointer"
            >
              <BookOpen className="w-4 h-4" />
              <span>Ver Guia dos 15 Polos</span>
            </button>
          </div>
        </div>

        <StatsCards linhas={linhas} totalArquivos={totalArquivos} />

        <DropZone
          onArquivosSelecionados={processarArquivos}
          onCarregarExemplos={() => {}}
          onAbrirModalPaste={() => setModalPasteAberto(true)}
          isProcessing={isProcessing}
          statusProcessamento={statusProcessamento}
        />

        <ConsolidatedTable
          linhas={linhas}
          onEditarLinha={handleEditarLinha}
          onExcluirLinha={handleExcluirLinha}
          onDuplicarLinha={handleDuplicarLinha}
          onAdicionarLinha={() => {
            setLinhaEmEdicao(null);
            setModalEditAberto(true);
          }}
          onLimparTudo={() => setConfirmarLimpezaAberto(true)}
          onExportarCSV={handleExportarCSV}
          onExportarExcel={handleExportarExcel}
          onCopiarClipboard={handleCopiarClipboard}
          referenciaMes={referenciaMes}
        />

      </main>

      <footer className="bg-white border-t border-slate-200 py-4 text-center text-xs text-slate-500">
        <div className="max-w-7xl mx-auto px-4 flex flex-col sm:flex-row items-center justify-between gap-2">
          <span>
            Consolidação de Relatórios ProLEEI RS • Formação Leitura e Escrita na Educação Infantil
          </span>
          <span className="text-[11px] text-slate-400">
            OCR nativo e regras SEDUC RS integradas
          </span>
        </div>
      </footer>

      <PoloExplorerModal
        isOpen={modalPolosAberto}
        onClose={() => setModalPolosAberto(false)}
        onSelecionarMunicipio={() => {
          setModalPolosAberto(false);
          setLinhaEmEdicao(null);
          setModalEditAberto(true);
        }}
      />

      <ManualPasteModal
        isOpen={modalPasteAberto}
        onClose={() => setModalPasteAberto(false)}
        onAdicionarLinhas={handleAdicionarLinhasManuais}
      />

      <EditRowModal
        isOpen={modalEditAberto}
        onClose={() => {
          setModalEditAberto(false);
          setLinhaEmEdicao(null);
        }}
        onSalvar={handleSalvarLinha}
        linhaInicial={linhaEmEdicao}
      />

      {confirmarLimpezaAberto && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/60 backdrop-blur-xs">
          <div className="bg-white rounded-2xl shadow-xl w-full max-w-md p-6 border border-slate-200">
            <h3 className="text-base font-bold text-slate-900 mb-2">
              Deseja limpar todos os registros?
            </h3>
            <p className="text-xs text-slate-500 mb-5">
              Esta ação removerá todos os {linhas.length} encontros atualmente consolidados na tabela.
            </p>
            <div className="flex justify-end gap-2">
              <button
                onClick={() => setConfirmarLimpezaAberto(false)}
                className="px-4 py-2 text-xs font-semibold text-slate-600 hover:bg-slate-100 rounded-lg transition-colors cursor-pointer"
              >
                Cancelar
              </button>
              <button
                onClick={handleLimparTudoConfirmado}
                className="px-4 py-2 text-xs font-semibold text-white bg-rose-600 hover:bg-rose-700 rounded-lg shadow-sm transition-colors cursor-pointer"
              >
                Confirmar e Limpar
              </button>
            </div>
          </div>
        </div>
      )}

    </div>
  );
}
```
